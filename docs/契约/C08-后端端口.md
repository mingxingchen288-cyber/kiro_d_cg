```python
class UnitOfWork(Protocol):

    def commit(self) -> None: ...
    def rollback(self) -> None: ...

class UnitOfWorkFactory(Protocol):

class SchemaInspector(Protocol):
    def columns(self, table: str) -> dict[str, ColumnInfo] | None: ...   # 表或视图的列，不存在时为 None；键为列名
    def object_type(self, name: str) -> Literal["table", "view"] | None: ...   # 是表还是视图，都不是时为 None；审批的锁定对象只能是表（C06 第 10 节）

@dataclass(frozen=True)
class ColumnInfo:
    name: str
    data_type: str                 # Oracle 的数据类型，例如 NUMBER、VARCHAR2、DATE
    nullable: bool
    length: int | None             # 字符列的长度
    precision: int | None          # 数值列的精度
    scale: int | None              # 数值列的小数位数，保存时据此处理小数位（write-pipeline WP-15）

class ConnectionFactory(Protocol):
    def legacy_readonly(self) -> ContextManager[sqlalchemy.engine.Connection]: ...   # 用 d_cg 账号开启只读事务（SET TRANSACTION READ ONLY），以 CMX.表名 读取旧库；只供 migration 和复制业务表使用
```

```python
    def rules_pack(self, pack_id: str) -> PackHandle: ...            # 已经 load_pack 的句柄

class ConfigSnapshotReader(Protocol):

    # 一次 Redis 读取（含当前代号和待生效登记）；只有待生效登记涉及本次要用的配置项、缺项或 Redis 不可用时才查 Oracle；Oracle 也不可用时抛 CONFIG_UNAVAILABLE

class ConfigImporter(Protocol):
    def import_items(self, items: list[ConfigDraft]) -> ImportReport: ...
    # ConfigDraft 为还没有保存的配置项（类型、编码、名称、内容），与 C09 校验接口相同；与前端保存使用同一套校验
```

- 快照中缺少要取的配置项时（例如 Redis 没有命中），先按 config 的 CF-06 回查 Oracle 并补写，不直接报错。回查后仍取不到时：
  - `page_view`：页面不存在为 404 NOT_FOUND，该角色不能访问这个页面为 403 PERMISSION_DENIED，与 C09 的页面视图接口一致；
  - `entity`、`operation`、`approval_flow`、`rules_pack`：说明已发布配置的内部引用断了（保存时已校验引用），为 500 INTERNAL_ERROR。
- `read` 回查后确认该角色的角色配置已被删除时，抛 401 SESSION_ROLE_REVOKED，由用户重新选择有效的角色。已经取得的快照不受影响，持有它的请求和命令按它执行到结束（C04）。
- `import_items` 整批校验：有一项不通过时抛 422 VALIDATION_FAILED，明细列出每个不通过的项，什么都不写。
  - 同一批中有两项或更多项类型和编码相同时，按不通过处理，明细列出其中每一项：无法判断该保留哪一项，只取一项会让其余的悄悄不生效。这项检查在跳过已有配置项之前进行，重复的项与已有配置项相同时也拒绝。
  - 与已有配置项类型和编码相同的项不覆盖，列入 `skipped`，既不写入也不参与校验；其余各项与已有配置、本批的其他项一起校验，引用可以指向两者。
  - 错误明细的 `path`（C03 第 1 节）以这一项在本批中的位置 `items[<序号>]` 开头，序号从 0 起，内容中的位置接在后面，例如 `items[3].levels[0].entries[1].approvers`；针对整项的错误（例如同批重复）只有 `items[<序号>]`。调用方据此找回这一项的来源（migration MG-11）。

## 6. 管道阶段

write-pipeline 按[架构第 7 节](../架构与组装.md)的顺序调用已注册的阶段。阶段之间只通过 `CommandState` 传递数据。

```python
class PipelineStage(Protocol):
    name: str
    def applies(self, state: CommandState, ctx: ExecutionContext) -> bool: ...
    def run(self, uow: UnitOfWork, state: CommandState, ctx: ExecutionContext) -> None: ...   # 失败时抛 PlatformError

@dataclass
class CommandState:                              # 除 command 外各字段在命令开始时为空，outcome 为 None
    command: Command
    involved: list[DocumentRef]
    current: dict[DocumentRef, DocumentData]     # 加锁后读到的当前值
    actions: list[ActionStep]                    # 要执行的动作：C06 第 6 节的通用动作（afterApproval 或操作的 actions）
    intents: list[RowIntent]                     # 动作展开出的修改，write-pipeline 据此加锁、合并（WP-18）
    candidate: dict[DocumentRef, DocumentData]   # 合并修改并计算后的数据
    effects: list[RuleEffect]                    # C07 effects[] 的项
    changes: ChangePlan                          # 只含有变化的列
    approvals: list[ApprovalState]               # 本命令提交或处理的审批；提交时每张提交审批的单据一个
    notifications: list[NotificationIntent]
    delivery_ids: list[str]
    prompts: list[Prompt]                        # C02 命令结果 prompts[] 的项
    skipped: list[SkippedRow]                    # C02 命令结果 skipped[] 的项
    pending_documents: set[DocumentRef]          # 提交审批的单据：本命令对这些单据的修改都不写业务表（approval AP-01）
    outcome: Literal["saved", "unchanged", "approvalSubmitted", "approvalProcessed"] | None   # 写入命令记录前由 write-pipeline 或触发审批阶段给出

@dataclass(frozen=True)
class RowIntent:                   # 动作要做的一行修改，还没有和当前数据合并
    document: DocumentRef          # 目标行所属的单据（明细行为它的主单）
    entity: str
    op: Literal["add", "modify", "delete"]
    row_key: str | None            # 新增行为 None
    client_row_id: str | None
    values: dict[str, Value]       # 新增、修改：要写的字段和值；删除：空

@dataclass
class ChangePlan:
    rows: list[RowChange]

@dataclass
class RowChange:
    document: DocumentRef          # 所属单据；新增的单据在取得主键前，record_key 为主表行的 clientRowId
    entity: str
    op: Literal["add", "modify", "delete"]
    row_key: str | None            # 新增行在取得主键前为 None
    client_row_id: str | None
    values: dict[str, Value]       # 新增：要写入的字段；修改：有变化的字段的新值；删除：空
    old: dict[str, Value]          # 修改：这些字段的旧值；删除：删除前的整行（审计用）；新增：空
```

| 阶段 | 实现模块 | 读取 | 写入 |
|---|---|---|---|
| 操作权限 | authz | `command`、ctx | 无；不通过时抛 PERMISSION_DENIED |
| 确定涉及的单据 | write-pipeline、operations、approval | `command` | `involved`、`actions` |
| 冻结检查 | approval | `involved`、`command`、ctx | 无；命中时抛 FROZEN_BY_APPROVAL，ctx 来源为本审批时放行本审批的冻结 |
| 审批处理 | approval | `current`、`command` | `approvals`、`actions`、`notifications` |
| 动作执行 | operations、write-pipeline、authz、rules-core | `current`、`actions`、`intents` | `intents`、`candidate`、`effects`、`changes`、`prompts`、`skipped` |
| 触发审批 | approval | `effects`、`candidate`、`changes` | `approvals`、`pending_documents`、`outcome`、`notifications` |
| 登记通知 | delivery | `notifications` | `delivery_ids` |
| 投递 | delivery | `delivery_ids` | 无；提交后执行 |

判重、加锁并读取、写入变更、审计、写入命令记录并提交，由 write-pipeline 自己完成，不是可注册的阶段。

动作执行每个动作一轮：operations 把动作展开为 `intents`（不读写 `candidate`，也不加锁）；write-pipeline 取出 `intents`，其中所属单据不在 `involved` 中的，先把这些单据加入 `involved`、加锁读入 `current`，再对它们执行一次冻结检查阶段，通过后才把 `intents` 合并进 `candidate` 并清空 `intents`，然后检查写权限、计算（write-pipeline WP-18）。

```python
class PermissionChecker(Protocol):

    def row_condition(self, ctx: ExecutionContext, page: str, entity: str) -> SqlCondition | None: ...   # 没有行权限时为 None
```

`RelatedQuery.find` 的查找和来源行：

- 按关联键查找时，用 `source` 中 `sourceKey` 所指实体（主表或某个明细）的全部行的键值查目标。按关联键找到的对象 `missing` 总为 False。
- `finder.entity` 为明细实体时（例如 C06 7.7 按关联键找原辅料明细），找到的每一行明细为一个对象：`record_key` 为它所属主单的主键（按实体的 `details.parentKey`），`row_keys` 为这行明细的主键。冻结记录、加锁和冻结检查都按所属主单（C06 7.3、WP-18）。
- 关联 SQL 的结果缺少 `recordKeyColumn` 列，或缺少 `rowKeyColumns` 中的列时，抛 500 INTERNAL_ERROR，整个查找失败，不返回部分结果：这是配置错误，无法判断要处理哪个对象。config 保存时已检查这些列（C06 第 10 节），运行时仍然缺少，说明保存之后表、视图或 SQL 引用的对象变了；不能把它当成"记录已不存在"跳过，否则配置错误会悄悄让同步失效。
- 关联 SQL 查出的某一行，`recordKeyColumn` 或 `rowKeyColumns` 中某列的值为空时，这一行表示的记录在查找时就已经不存在，例如成本单的明细还在，它所属的成本单已被删除。这一行照常返回：`missing` 为 True，`record_key` 为查到的单据主键（为空时为 None），`row_keys` 为空，`display` 照常取值；其余各行不受影响，也不报错。调用方不冻结、不更新 `missing` 的对象；approval 把它保存为候选对象，`selection` 为 `approverSelects` 的审批流在审批详情中显示为"记录已不存在"，`selection` 为 `all` 的只记入执行结果（approval AP-21）。
- 按钮操作中的 `updateRecords`：operations 传入选中单据加锁后从数据库读到的当前数据，加上本命令前面各动作计算后的结果（`CommandState.current`、`candidate`），不用前端的值（C02 第 1 节）。按钮命令不带修改，关联键的 `rows` 只能为 `all`（C06 第 10 节）。
- 审批流中的查找（`targets.find`、`freeze`，以及 `afterApproval` 中的 `updateRecords`）：调用方（approval、operations）按查找方式的 `rows`（C06 7.2）准备 `source`，不另查数据库：
  - `rows` 为 `all`：传入合并并计算后的完整单据。
  - `rows` 为 `changed`：另建一个只供这次查找的副本，原单据不变，随后的计算和保存仍用完整的单据。按 `CommandState.changes`（第 6 节的 `ChangePlan`）选出 `sourceKey` 所指实体中有实际变化的行：新增、修改的行取合并计算后的有效值（`candidate` 中的整行）；删除的行只在配置监听删除时（审批流的 `trigger.operations` 含 `delete`）作为来源行，取删除前的整行（`RowChange.old`）。
  - `sourceKey` 指向明细时，副本的这个明细只放上面选出的行，主表行照常给出，作为查询的上下文。`sourceKey` 指向主表而主表没有实际变化时，没有来源行，不调用 `find`，结果为空，不能因为主表照常给出就把它当成变化的来源。
  - 修改了关联键的行只按新值查找目标，不同时处理旧值对应的目标；需要处理旧关系时，由业务动作明确规定。
  - 例如一张单据有 10 条明细，只改了第 3 条：整单计算仍用 10 条明细；`rows` 为 `changed` 的关联查找只用第 3 条，查找用的副本不影响随后的计算和保存。
